# PAKLib (C#) public API -> PAKLibC (namespace PAKLib, PAKLibC\PAK.h). One line per public type and member;
# check_api.py fails on anything unmapped, stale or naming a C++ name PAK.h lacks. Overloads: Member(n) = n params.
# "(none)" marks a member with no C++ counterpart, with the reason. "extra:" lines are the allowed C++-only API.

PAK                                  -> pak (PAK and PAKData are one struct)
PAK.ctor                             -> pak::pak() (value-initialised: pak{})
PAK.Data                             -> (none) the pak itself holds the data
PAK.ReadFromFile                     -> loadpak(path)
PAK.SaveToFile                       -> save_to_file(pak, path)
PAK.RepairFile                       -> repair_file(input, output, force)

PAKData                              -> pak
PAKData.ctor                         -> pak::pak() (value-initialised: pak{})
PAKData.Read                         -> pak::read(bytes)
PAKData.ToBytes                      -> pak::to_bytes()
PAKData.Write                        -> pak::write(path)
PAKData.ConvertImages                -> pak::convert_images(format)
PAKData.Sprites                      -> pak::sprites
PAKData.SpriteCount                  -> pak::sprites.size() (pak::sprite_count is the count as read, an extra)

# Allowed C++-only API (rulings: the fast readers and get_sprite; R2: the headers and the count as read)
extra: get_sprite(path, index)       one sprite with PAK.ReadFromFile's path checks and PAKData.Read's per-sprite checks
extra: loadpak_fast(path)            lenient reader: no path, magic or image checks; the same count and bounds rules
extra: get_sprite_fast(path, index)  lenient single sprite: as loadpak_fast, plus the index check
extra: pak::header                   the file header as read
extra: pak::sprite_count             the sprite count as read
extra: sprite::header                each sprite's 100-byte header as read
